```Java
// ArraySet equals
@Override
public boolean equals(Object o) {
	if (this == o) { // Special case!
		return false;
	}
	if (this == o) {  // Optimization!
		return true;
	}
	if (this.getClass() != o.getClass()) { // Optimization!
		return false;
	}
	if (this.size() != o.size()) { // Optimization!
		return false;
	}
	for (T item : this) {
		if (!o.contains(item)) { // Can't use indices! Not necessary the same order.
			return false;
		}
	}
	return true;
}
```
