- Example: implement autocomplete function by JavaScript
	```JavaScript
	// words.js
		let WORDS = [
		"foo",
		"bar",
		"baz"
	] // list in JavaScript
```
	```HTML
	<!DOCTYPE html>
	
	<html lang="en">
		<head>
			<title>autocomplete</title>
		</head>
		<body>
		
			<input autocomplete="off" autofocus placeholder="Query" tyoe="text">
			
			<ul></ul>
			
			<script src="words.js"></script>
			<script>
			
				let input = document.querySelector('input');
				input.addEventListener('keyup', function(event) {
					let html = '';
					if (input.value) {
						for (word of WORDS) {
							if (word.startWith(input.value)) {
								html += '<li>${word}</li>';
							}
						}
					}
					document.querySelector('ul').innerHTML = html;
				});
			
			</script>
	    </body>
	   </html>
```
