	import sqlite3
	
	# Connect to the SQLite database
	conn = sqlite3.connect('favorites.db')
	cursor = conn.cursor()
	
	# Prompt the user for their favorite language
	favorite = input("Favorite: ")
	
	# Execute the query with a parameterized input to prevent SQL injection
	cursor.execute("SELECT COUNT(*) AS n FROM favorites WHERE language = ?", (favorite,))
	
	# Fetch the result
	row = cursor.fetchone()
	
	# Check if any row is returned and print the count
	if row:
	    print(row[0])
	else:
	    print("No such language found.")
	
	# Close the connection
	conn.close()
